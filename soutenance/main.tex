% Insère une figure du dépôt si elle existe. Le fallback permet de compiler
% le fichier même lorsque le dossier soutenance est copié seul.
\newcommand{\repofigure}[2][0.62\textheight]{%
  \IfFileExists{#2}{%
    \includegraphics[width=\linewidth,height=#1,keepaspectratio]{#2}%
  }{%
    \IfFileExists{../#2}{%
      \includegraphics[width=\linewidth,height=#1,keepaspectratio]{../#2}%
    }{%
      \fbox{\begin{minipage}[c][#1][c]{0.92\linewidth}\centering\scriptsize\color{Muted}
        Figure attendue :\\[0.25em]\texttt{\detokenize{#2}}
      \end{minipage}}%
    }%
  }%
}
