\small Contrôleurs → Services → Repositories\\DTO + mappers pour isoler le contrat HTTP de la persistance.

\begin{columns}[T]
\begin{column}{0.49\linewidth}
\begin{block}{Pourquoi trois métriques ?}
\small MAE = erreur métier moyenne · RMSE pénalise les gros écarts · R² mesure la variance expliquée.
\end{block}
\end{column}
\begin{column}{0.49\linewidth}
\begin{block}{Ce que SHAP ne dit pas}
\small Contribution statistique à une prédiction \warn{≠ causalité métier}.
\end{block}
\end{column}

\begin{frame}{Et si la volumétrie change : ce que je n'ai pas fait}
\eyebrow{C2.5 · Architecture cible}\competency{C2.5}
\begin{columns}[T]
\begin{column}{0.55\linewidth}
\centering
\repofigure[0.48\textheight]{figures/ia/nouvelle_architecture_simplifie.png}
\end{column}
\begin{column}{0.41\linewidth}
\begin{block}{Projet réalisé}
\small Jeu de données modéré, calcul local. \strong{Pas de traitement distribué.}
\end{block}
\small
\begin{enumerate}
  \item Séparer ingestion / brut / exploitation
  \item Distribuer seulement si nécessaire
  \item Versionner et surveiller le modèle
  \item Intégrer la sensibilité des données dès l'ingestion
\end{enumerate}
\vspace{0.2em}
\small\strong{Sur-dimensionner aurait été une mauvaise décision d'ingénierie — et de numérique responsable.}
\end{column}
\end{columns}
\end{frame}
