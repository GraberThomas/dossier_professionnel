\section{Conclusion}

\begin{frame}{Ce que l'ensemble démontre}
\eyebrow{En conclusion}
\begin{columns}[T]
\begin{column}{0.24\linewidth}
\begin{block}{Analyser}
\small Reconstituer un système complexe avant de le modifier et distinguer la demande du problème réel.
\end{block}
\end{column}
\begin{column}{0.24\linewidth}
\begin{block}{Concevoir}
\small Choisir un niveau d'abstraction et justifier le compromis plutôt que suivre une tendance.
\end{block}
\end{column}
\begin{column}{0.24\linewidth}
\begin{block}{Industrialiser}
\small Versionner, valider, automatiser, surveiller et documenter selon le contexte.
\end{block}
\end{column}
\begin{column}{0.24\linewidth}
\begin{block}{Piloter}
\small Tenir une trajectoire quand les hypothèses changent et rendre les arbitrages visibles.
\end{block}
\end{column}
\end{columns}
\vspace{0.75em}
\begin{block}{Ce que j'ai surtout appris}
\centering\small Une bonne décision technique tient compte du \strong{risque}, de la \strong{continuité d'activité}, des \strong{compétences disponibles} et du \strong{coût de remplacement des dépendances existantes} — pas seulement de l'élégance de la solution.
\end{block}
